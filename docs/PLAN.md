### 1.2 The forgery generator — `tools/forge.py`

Six forgery modes, spanning *traditional* and *AI-era* attacks. Modes 1–5 need no API key.

```python
"""
Generate a labelled forgery corpus with pixel-precise masks.

  python tools/forge.py --src data/authentic --out data/forged --n 60

Modes
  1 digit_splice   copy a digit from elsewhere in the SAME doc over a target digit
  2 digit_retype   render a new digit in a matched font over the target
  3 copy_move      duplicate a region (stamp, signature, line item)
  4 resave         re-encode a patch at a different JPEG quality  (compression-history attack)
  5 blur_erase     content-aware-ish removal of a field
  6 ai_inpaint     OPTIONAL: call an image-edit API   ← the modern threat
"""
import argparse, json, pathlib, random, io, hashlib
import cv2, numpy as np
from PIL import Image
import pytesseract
from pytesseract import Output


# ---------- find the numeric fields worth attacking ----------
def find_numeric_boxes(img_bgr, min_h=14):
    gray = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    d = pytesseract.image_to_data(gray, output_type=Output.DICT)
    boxes = []
    for i, txt in enumerate(d["text"]):
        t = txt.strip().replace(",", "").replace(".", "")
        if t.isdigit() and len(t) >= 3 and int(d["conf"][i]) > 40 and d["height"][i] >= min_h:
            boxes.append({"bbox": [d["left"][i], d["top"][i],
                                   d["left"][i] + d["width"][i], d["top"][i] + d["height"][i]],
                          "text": txt.strip()})
    return boxes


# ---------- mode 1: splice a digit from elsewhere in the same document ----------
def digit_splice(img, boxes, rng):
    if len(boxes) < 2: return None
    tgt, src = rng.sample(boxes, 2)
    x1, y1, x2, y2 = tgt["bbox"]
    dw = (x2 - x1) // max(len(tgt["text"]), 1)          # approx one glyph
    sx1, sy1, sx2, sy2 = src["bbox"]
    sw = (sx2 - sx1) // max(len(src["text"]), 1)
    patch = img[sy1:sy2, sx1:sx1 + sw]
    if patch.size == 0 or dw <= 2: return None
    patch = cv2.resize(patch, (dw, y2 - y1))
    out = img.copy()
    out[y1:y2, x1:x1 + dw] = patch                       # replace the LEADING digit
    mask = np.zeros(img.shape[:2], np.uint8); mask[y1:y2, x1:x1 + dw] = 255
    return out, mask, {"mode": "digit_splice", "bbox": [x1, y1, x1 + dw, y2],
                       "orig": tgt["text"]}


# ---------- mode 2: retype a digit ----------
def digit_retype(img, boxes, rng):
    if not boxes: return None
    tgt = rng.choice(boxes)
    x1, y1, x2, y2 = tgt["bbox"]
    dw = (x2 - x1) // max(len(tgt["text"]), 1)
    if dw <= 2: return None
    out = img.copy()
    roi = out[y1:y2, x1:x1 + dw]
    bg = int(np.median(out[max(0, y1-4):y1, x1:x1 + dw]))          # sample paper colour
    fg = int(np.percentile(cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY), 5))
    out[y1:y2, x1:x1 + dw] = bg
    newd = rng.choice("123456789")
    scale = (y2 - y1) / 22.0
    cv2.putText(out, newd, (x1 + 1, y2 - 2), cv2.FONT_HERSHEY_SIMPLEX,
                scale, (fg, fg, fg), max(1, int(scale * 1.6)), cv2.LINE_AA)
    mask = np.zeros(img.shape[:2], np.uint8); mask[y1:y2, x1:x1 + dw] = 255
    return out, mask, {"mode": "digit_retype", "bbox": [x1, y1, x1 + dw, y2],
                       "orig": tgt["text"], "new_digit": newd}


# ---------- mode 3: copy-move a whole region ----------
def copy_move(img, boxes, rng):
    h, w = img.shape[:2]
    bw, bh = rng.randint(w // 12, w // 6), rng.randint(h // 26, h // 12)
    sx, sy = rng.randint(0, w - bw - 1), rng.randint(0, h - bh - 1)
    dx, dy = rng.randint(0, w - bw - 1), rng.randint(0, h - bh - 1)
    if abs(sx - dx) < bw and abs(sy - dy) < bh: return None
    out = img.copy()
    out[dy:dy + bh, dx:dx + bw] = img[sy:sy + bh, sx:sx + bw]
    mask = np.zeros((h, w), np.uint8); mask[dy:dy + bh, dx:dx + bw] = 255
    return out, mask, {"mode": "copy_move", "bbox": [dx, dy, dx + bw, dy + bh],
                       "src_bbox": [sx, sy, sx + bw, sy + bh]}


# ---------- mode 4: recompress a patch (compression-history attack) ----------
def resave_patch(img, boxes, rng):
    if not boxes: return None
    tgt = rng.choice(boxes)
    x1, y1, x2, y2 = tgt["bbox"]
    pad = 6
    x1, y1 = max(0, x1 - pad), max(0, y1 - pad)
    x2, y2 = min(img.shape[1], x2 + pad), min(img.shape[0], y2 + pad)
    patch = img[y1:y2, x1:x2]
    if patch.size == 0: return None
    q = rng.choice([55, 65, 98])                       # deliberately mismatched quality
    ok, enc = cv2.imencode(".jpg", patch, [cv2.IMWRITE_JPEG_QUALITY, q])
    out = img.copy(); out[y1:y2, x1:x2] = cv2.imdecode(enc, cv2.IMREAD_COLOR)
    mask = np.zeros(img.shape[:2], np.uint8); mask[y1:y2, x1:x2] = 255
    return out, mask, {"mode": "resave", "bbox": [x1, y1, x2, y2], "quality": q}


# ---------- mode 5: erase a field ----------
def blur_erase(img, boxes, rng):
    if not boxes: return None
    tgt = rng.choice(boxes)
    x1, y1, x2, y2 = tgt["bbox"]
    out = img.copy()
    m = np.zeros(img.shape[:2], np.uint8); m[y1:y2, x1:x2] = 255
    out = cv2.inpaint(out, m, 3, cv2.INPAINT_TELEA)
    return out, m, {"mode": "blur_erase", "bbox": [x1, y1, x2, y2]}


MODES = {"digit_splice": digit_splice, "digit_retype": digit_retype,
         "copy_move": copy_move, "resave": resave_patch, "blur_erase": blur_erase}


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--src", required=True); ap.add_argument("--out", required=True)
    ap.add_argument("--n", type=int, default=60); ap.add_argument("--seed", type=int, default=7)
    a = ap.parse_args()
    rng = random.Random(a.seed)
    src = sorted(p for p in pathlib.Path(a.src).glob("*")
                 if p.suffix.lower() in (".jpg", ".jpeg", ".png"))
    out = pathlib.Path(a.out); (out / "masks").mkdir(parents=True, exist_ok=True)
    meta, made, tries = [], 0, 0

    while made < a.n and tries < a.n * 12:
        tries += 1
        p = rng.choice(src)
        img = cv2.imread(str(p))
        if img is None: continue
        boxes = find_numeric_boxes(img)
        name = rng.choice(list(MODES))
        r = MODES[name](img, boxes, rng)
        if r is None: continue
        forged, mask, info = r
        stem = f"F{made:04d}_{name}"
        fp = out / f"{stem}.jpg"
        # final save at a realistic quality -> creates the double-compression signature
        cv2.imwrite(str(fp), forged, [cv2.IMWRITE_JPEG_QUALITY, rng.choice([85, 90, 92])])
        cv2.imwrite(str(out / "masks" / f"{stem}.png"), mask)
        info.update(file=fp.name, source=p.name, label="forged",
                    mask=f"masks/{stem}.png",
                    tampered_pixel_frac=round(float((mask > 0).mean()), 5))
        meta.append(info); made += 1

    (out / "manifest.json").write_text(json.dumps(meta, indent=2))
    fr = [m["tampered_pixel_frac"] for m in meta]
    print(f"\n{made} forgeries -> {out}")
    print(f"tampered pixel fraction: min {min(fr):.4f}  median {np.median(fr):.4f}  max {max(fr):.4f}")
    print("(DOCFORGE-BENCH reports 0.0027-0.0417 in the wild — check you are in range)")


if __name__ == "__main__":
    main()
```

```python
"""Core types. An EvidenceCard is IMMUTABLE and content-addressed."""
from __future__ import annotations
import hashlib, json, time
from dataclasses import dataclass, field, asdict
from typing import Any


def sha256_bytes(b: bytes) -> str:
    return hashlib.sha256(b).hexdigest()


@dataclass(frozen=True)
class EvidenceCard:
    """One probe result. The hash commits to the EXACT pixel bytes examined."""
    id: str
    probe: str
    params: dict
    bbox: list | None
    px_on_target: int
    crop_sha256: str            # sha256 of the PNG-encoded crop -> the bond anchor
    crop_path: str | None
    observation: str            # short, factual, machine-generated
    numeric: dict               # z-scores, ratios etc. -- what the Adjudicator reasons over
    cost_units: float
    ts: float = field(default_factory=time.time)

    def manifest_line(self) -> str:
        """The ONLY representation the starved Adjudicator ever sees."""
        n = " ".join(f"{k}={v}" for k, v in self.numeric.items())
        return (f"{self.id}  {self.probe:<14} px={self.px_on_target:<6} "
                f"sha={self.crop_sha256[:8]}  {n}  | {self.observation}")


@dataclass
class Claim:
    id: str
    text: str
    bond: list[str]             # evidence card ids
    bond_status: str = "untested"   # untested | holds | broken
    bond_detail: dict = field(default_factory=dict)


@dataclass
class Verdict:
    claim_id: str
    verdict: str                # tamper_detected | authentic | synthetic_evidence
                                # | insufficient_evidence | undetermined
    confidence: float | None
    claims: list[Claim]
    controls: dict
    cost: dict
    latency_s: float
    evidence_chain: list[str]

    def to_dict(self):
        d = asdict(self); d["claims"] = [asdict(c) for c in self.claims]; return d

    @property
    def bonded_claims(self):
        return [c for c in self.claims if c.bond_status == "holds"]

    @property
    def is_bonded(self):
        """A verdict only counts if it has at least one surviving bonded claim."""
        return len(self.bonded_claims) > 0
```

`verdict/ledger.py`:

```python
"""Append-only, content-addressed evidence store. Provenance cannot be faked."""
import json, pathlib, sqlite3, time
import cv2
from .types import EvidenceCard, sha256_bytes


class Ledger:
    def __init__(self, root="data", db="data/ledger.db"):
        self.root = pathlib.Path(root); (self.root / "crops").mkdir(parents=True, exist_ok=True)
        self.db = sqlite3.connect(db, check_same_thread=False)
        self.db.execute("""CREATE TABLE IF NOT EXISTS cards(
            sha TEXT PRIMARY KEY, claim_id TEXT, card_id TEXT, probe TEXT,
            params TEXT, bbox TEXT, px INTEGER, obs TEXT, numeric TEXT,
            crop_path TEXT, cost REAL, ts REAL)""")
        self.db.execute("""CREATE TABLE IF NOT EXISTS claims(
            claim_id TEXT PRIMARY KEY, phash TEXT, verdict TEXT, ts REAL)""")
        self.db.commit()
        self._cache: dict[str, EvidenceCard] = {}

    def add(self, claim_id: str, card_id: str, probe: str, params: dict,
            bbox, crop_bgr, observation: str, numeric: dict, cost: float) -> EvidenceCard:
        ok, buf = cv2.imencode(".png", crop_bgr)      # PNG = lossless = stable hash
        assert ok, "crop encode failed"
        sha = sha256_bytes(buf.tobytes())
        path = self.root / "crops" / f"{sha[:16]}.png"
        if not path.exists():
            path.write_bytes(buf.tobytes())
        card = EvidenceCard(id=card_id, probe=probe, params=params, bbox=bbox,
                            px_on_target=int(crop_bgr.shape[0]),
                            crop_sha256=sha, crop_path=str(path),
                            observation=observation, numeric=numeric, cost_units=cost)
        self.db.execute("INSERT OR IGNORE INTO cards VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
                        (sha, claim_id, card_id, probe, json.dumps(params),
                         json.dumps(bbox), card.px_on_target, observation,
                         json.dumps(numeric), str(path), cost, card.ts))
        self.db.commit()
        self._cache[card_id] = card
        return card

    def get(self, card_id: str) -> EvidenceCard | None:
        return self._cache.get(card_id)

    def verify_provenance(self, card_id: str) -> bool:
        """A card must exist AND its bytes must still hash to its recorded sha."""
        c = self._cache.get(card_id)
        if c is None or not c.crop_path:
            return False
        return sha256_bytes(pathlib.Path(c.crop_path).read_bytes()) == c.crop_sha256

    def manifest(self, ids=None) -> str:
        cards = [c for k, c in self._cache.items() if ids is None or k in ids]
        return "\n".join(c.manifest_line() for c in sorted(cards, key=lambda c: c.id))
```

## 3. The probes — the meat

`verdict/probes/cheap.py` — **zero VLM calls, run first, resolve ~40% of cases:**

```python
"""Deterministic probes. No model. No hallucination. Nearly free."""
import re, io, pathlib
import numpy as np, cv2
from PIL import Image
import piexif


def exif_audit(path: str) -> dict:
    out = {"has_exif": False, "make": None, "model": None, "software": None,
           "datetime": None, "flags": []}
    try:
        ex = piexif.load(str(path))
        z = ex.get("0th", {}) or {}
        g = lambda t: z.get(t, b"").decode(errors="ignore") or None
        out.update(has_exif=bool(z), make=g(piexif.ImageIFD.Make),
                   model=g(piexif.ImageIFD.Model), software=g(piexif.ImageIFD.Software),
                   datetime=g(piexif.ImageIFD.DateTime))
    except Exception:
        pass
    if not out["has_exif"]:
        out["flags"].append("no_exif")                       # screenshot / re-saved / generated
    sw = (out["software"] or "").lower()
    if any(k in sw for k in ("photoshop", "gimp", "canva", "snapseed", "picsart", "ai")):
        out["flags"].append(f"editor_tag:{out['software']}")
    return out


def phash(img_bgr, hash_size=8) -> str:
    g = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    g = cv2.resize(g, (32, 32), interpolation=cv2.INTER_AREA).astype(np.float32)
    d = cv2.dct(g)[:hash_size, :hash_size].flatten()
    med = np.median(d[1:])                                    # drop DC
    return "".join("1" if v > med else "0" for v in d)


def hamming(a: str, b: str) -> int:
    return sum(x != y for x, y in zip(a, b))


NUM = re.compile(r"(\d[\d,]*\.?\d{0,2})")

def cross_field(ocr_text: str) -> dict:
    """
    Cheapest, sharpest probe in the system: does the arithmetic close?
    Forgers change ONE number. They almost never recompute the bill.
    """
    lines = [l.strip() for l in ocr_text.splitlines() if l.strip()]
    total = None
    for l in lines:
        if re.search(r"\b(total|grand total|amount payable|net)\b", l, re.I):
            m = NUM.findall(l)
            if m:
                total = float(m[-1].replace(",", ""))
    items = []
    for l in lines:
        if re.search(r"\b(total|subtotal|tax|gst|cgst|sgst|discount|round)\b", l, re.I):
            continue
        m = NUM.findall(l)
        if m:
            try:
                v = float(m[-1].replace(",", ""))
                if 0 < v < 1e7: items.append(v)
            except ValueError:
                pass
    s = round(sum(items), 2)
    if total is None or not items:
        return {"status": "unparseable", "items_sum": s, "stated_total": total}
    rel = abs(s - total) / max(total, 1.0)
    return {"status": "mismatch" if rel > 0.02 else "consistent",
            "items_sum": s, "stated_total": total,
            "delta": round(total - s, 2), "rel_error": round(rel, 4)}
```

```python
"""Compression-history probes. This is where localized regeneration shows up."""
import io
import numpy as np, cv2
from PIL import Image


def ela(img_bgr, qualities=(70, 80, 90)) -> tuple[np.ndarray, dict]:
    """
    Multi-scale Error Level Analysis.
    arXiv 2607.06615: multi-scale gives +0.180 AUC over single-quality. Do not use one q.
    """
    maps = []
    for q in qualities:
        ok, enc = cv2.imencode(".jpg", img_bgr, [cv2.IMWRITE_JPEG_QUALITY, int(q)])
        dec = cv2.imdecode(enc, cv2.IMREAD_COLOR)
        maps.append(cv2.absdiff(img_bgr, dec).astype(np.float32).max(axis=2))
    stack = np.stack(maps)
    m = stack.mean(axis=0)
    return m, {"mean": float(m.mean()), "std": float(m.std()),
               "p99": float(np.percentile(m, 99)),
               "cross_q_ratio": float(maps[-1].mean() / max(maps[0].mean(), 1e-6))}


def ela_region_z(ela_map, bbox, doc_mean=None, doc_std=None) -> dict:
    """Region z-score. Thresholds come from CALIBRATION, never from a constant."""
    x1, y1, x2, y2 = [int(v) for v in bbox]
    r = ela_map[y1:y2, x1:x2]
    if r.size == 0: return {"z": 0.0, "region_mean": 0.0}
    mu = doc_mean if doc_mean is not None else float(ela_map.mean())
    sd = doc_std if doc_std is not None else float(ela_map.std())
    return {"z": round(float((r.mean() - mu) / max(sd, 1e-6)), 3),
            "region_mean": round(float(r.mean()), 3),
            "doc_mean": round(mu, 3), "doc_std": round(sd, 3)}


def jpeg_ghost(img_bgr, qualities=range(50, 100, 5), block=16):
    """
    JPEG ghost: a region originally compressed at q* shows a local MINIMUM in
    difference when recompressed at q*. A spliced/inpainted region has a
    different q* from the rest -> a visible 'ghost'.
    """
    h, w = img_bgr.shape[:2]
    diffs = []
    for q in qualities:
        ok, enc = cv2.imencode(".jpg", img_bgr, [cv2.IMWRITE_JPEG_QUALITY, int(q)])
        d = ((img_bgr.astype(np.float32) - cv2.imdecode(enc, cv2.IMREAD_COLOR).astype(np.float32)) ** 2).mean(axis=2)
        diffs.append(cv2.blur(d, (block, block)))
    D = np.stack(diffs)                                  # [Q, H, W]
    argmin_q = np.array(list(qualities))[D.argmin(axis=0)]
    med = float(np.median(argmin_q))
    outlier = (np.abs(argmin_q - med) > 12).astype(np.uint8) * 255
    return argmin_q, outlier, {"median_q": med,
                               "outlier_frac": round(float((outlier > 0).mean()), 5)}


def dct_quant(path: str) -> dict:
    """
    JPEG quantization tables. Per arXiv 2605.19688 (DocQT), qtable identity is a
    strong, under-used signal: real pipelines have a SMALL, STABLE table bank.
    """
    try:
        im = Image.open(path)
        qt = getattr(im, "quantization", None) or {}
        tabs = {str(k): list(v) for k, v in qt.items()}
        import hashlib
        h = hashlib.sha256(repr(sorted(tabs.items())).encode()).hexdigest()[:16]
        return {"has_qtable": bool(tabs), "qtable_hash": h,
                "n_tables": len(tabs),
                "luma_mean": round(float(np.mean(list(qt.values())[0])), 2) if qt else None}
    except Exception as e:
        return {"has_qtable": False, "error": str(e)}
```

```python
"""Noise-residual and copy-move probes."""
import numpy as np, cv2


def noise_residual(img_bgr, block=32):
    """
    High-pass sensor/render residual.
    FLAME (ICML 2026, arXiv 2606.02178) shows diffusion SUPPRESSES local
    high-frequency variance -> generated regions look ANOMALOUSLY SMOOTH.
    So we look for LOW variance, not high.
    """
    g = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    den = cv2.fastNlMeansDenoising(g, None, h=7, templateWindowSize=7, searchWindowSize=21)
    res = g.astype(np.float32) - den.astype(np.float32)
    H, W = res.shape
    gh, gw = H // block, W // block
    var = np.zeros((gh, gw), np.float32)
    for i in range(gh):
        for j in range(gw):
            var[i, j] = res[i*block:(i+1)*block, j*block:(j+1)*block].var()
    med = float(np.median(var))
    low = (var < 0.35 * med).astype(np.uint8) * 255       # suspiciously smooth
    return res, var, {"median_block_var": round(med, 4),
                      "low_var_frac": round(float((low > 0).mean()), 5),
                      "min_block_var": round(float(var.min()), 4)}


def region_hf_ratio(var_grid, bbox, img_shape, block=32) -> dict:
    H, W = img_shape[:2]
    x1, y1, x2, y2 = [int(v) for v in bbox]
    i1, i2 = max(0, y1 // block), min(var_grid.shape[0], y2 // block + 1)
    j1, j2 = max(0, x1 // block), min(var_grid.shape[1], x2 // block + 1)
    sub = var_grid[i1:i2, j1:j2]
    if sub.size == 0: return {"hf_ratio": 1.0}
    return {"hf_ratio": round(float(sub.mean() / max(np.median(var_grid), 1e-6)), 3)}


def copy_move(img_bgr, min_dist=40, ratio=0.75):
    """ORB self-matching. Catches duplicated stamps, signatures, line items."""
    g = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    orb = cv2.ORB_create(nfeatures=6000)
    kp, des = orb.detectAndCompute(g, None)
    if des is None or len(kp) < 20:
        return [], {"n_pairs": 0}
    bf = cv2.BFMatcher(cv2.NORM_HAMMING)
    knn = bf.knnMatch(des, des, k=3)
    pairs = []
    for ms in knn:
        for m in ms[1:]:                                  # skip self-match
            if ms[1].distance < ratio * ms[-1].distance:
                p, q = kp[m.queryIdx].pt, kp[m.trainIdx].pt
                if np.hypot(p[0]-q[0], p[1]-q[1]) > min_dist:
                    pairs.append((p, q))
                break
    return pairs, {"n_pairs": len(pairs),
                   "frac_of_kp": round(len(pairs) / max(len(kp), 1), 4)}
```

```python
"""Typography probes -- generators rarely reproduce baseline and kerning exactly."""
import numpy as np, cv2, pytesseract
from pytesseract import Output


def font_metrics(img_bgr, bbox=None) -> dict:
    g = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY)
    d = pytesseract.image_to_data(g, output_type=Output.DICT)
    words = [{"x": d["left"][i], "y": d["top"][i], "w": d["width"][i],
              "h": d["height"][i], "t": d["text"][i], "c": int(d["conf"][i]),
              "line": (d["block_num"][i], d["par_num"][i], d["line_num"][i])}
             for i in range(len(d["text"]))
             if d["text"][i].strip() and int(d["conf"][i]) > 40]
    if len(words) < 5:
        return {"status": "insufficient_text", "n_words": len(words)}

    # baseline jitter per text line
    lines, jitter = {}, []
    for w in words:
        lines.setdefault(w["line"], []).append(w)
    for ws in lines.values():
        if len(ws) < 2: continue
        b = np.array([w["y"] + w["h"] for w in ws], np.float32)
        jitter.append(float(b.std()))
    doc_jitter = float(np.median(jitter)) if jitter else 0.0

    def stroke_width(roi):
        if roi.size == 0: return 0.0
        _, bw = cv2.threshold(roi, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
        dt = cv2.distanceTransform(bw, cv2.DIST_L2, 3)
        v = dt[dt > 0]
        return float(2.0 * np.mean(v)) if v.size else 0.0

    sws = [stroke_width(g[w["y"]:w["y"]+w["h"], w["x"]:w["x"]+w["w"]]) for w in words]
    sws = [s for s in sws if s > 0]
    doc_sw = float(np.median(sws)) if sws else 0.0

    out = {"status": "ok", "n_words": len(words),
           "doc_baseline_jitter_px": round(doc_jitter, 3),
           "doc_stroke_width_px": round(doc_sw, 3)}

    if bbox:
        x1, y1, x2, y2 = [int(v) for v in bbox]
        inside = [w for w in words if w["x"] >= x1-4 and w["x"]+w["w"] <= x2+4
                  and w["y"] >= y1-4 and w["y"]+w["h"] <= y2+4]
        if inside:
            sw_r = np.median([s for s in
                              [stroke_width(g[w["y"]:w["y"]+w["h"], w["x"]:w["x"]+w["w"]])
                               for w in inside] if s > 0] or [doc_sw])
            same = [w for w in words if w["line"] == inside[0]["line"]]
            if len(same) >= 2:
                bl = np.array([w["y"] + w["h"] for w in same], np.float32)
                off = float(abs((inside[0]["y"] + inside[0]["h"]) - np.median(bl)))
            else:
                off = 0.0
            out.update(region_stroke_width_px=round(float(sw_r), 3),
                       stroke_ratio=round(float(sw_r / max(doc_sw, 1e-6)), 3),
                       baseline_offset_px=round(off, 3),
                       baseline_z=round(off / max(doc_jitter, 0.15), 3))
    return out
```

```python
"""Zoom = the workhorse. This is where the 45x information gain comes from."""
import numpy as np, cv2


def zoom(img_bgr, bbox, pad=0.10, min_side=448):
    """Native-resolution crop. NO downsampling -- that is the entire point."""
    H, W = img_bgr.shape[:2]
    x1, y1, x2, y2 = [int(v) for v in bbox]
    pw, ph = int((x2-x1)*pad), int((y2-y1)*pad)
    x1, y1 = max(0, x1-pw), max(0, y1-ph)
    x2, y2 = min(W, x2+pw), min(H, y2+ph)
    crop = img_bgr[y1:y2, x1:x2]
    if crop.size == 0: return None, {"px_on_target": 0}
    px = crop.shape[0]
    if max(crop.shape[:2]) < min_side:          # upscale ONLY for model ingestion
        s = min_side / max(crop.shape[:2])      # px_on_target still reports the TRUE value
        crop = cv2.resize(crop, None, fx=s, fy=s, interpolation=cv2.INTER_LANCZOS4)
    return crop, {"px_on_target": px, "bbox": [x1, y1, x2, y2],
                  "upscaled": max(crop.shape[:2]) != px}


def enhance(img_bgr, mode="clahe"):
    """
    Reveal what was CAPTURED but is not PERCEPTIBLE.
    NOTE: enhancement does NOT increase px_on_target. It is not evidence
    acquisition and must never be treated as such.
    """
    if mode == "clahe":
        lab = cv2.cvtColor(img_bgr, cv2.COLOR_BGR2LAB)
        lab[:, :, 0] = cv2.createCLAHE(3.0, (8, 8)).apply(lab[:, :, 0])
        return cv2.cvtColor(lab, cv2.COLOR_LAB2BGR)
    if mode == "gamma_lift":
        t = np.array([((i/255.0) ** 0.45) * 255 for i in range(256)]).astype("uint8")
        return cv2.LUT(img_bgr, t)
    if mode == "unsharp":
        return cv2.addWeighted(img_bgr, 1.6, cv2.GaussianBlur(img_bgr, (0, 0), 3), -0.6, 0)
    if mode == "edge":
        e = cv2.Canny(cv2.cvtColor(img_bgr, cv2.COLOR_BGR2GRAY), 60, 160)
        return cv2.cvtColor(e, cv2.COLOR_GRAY2BGR)
    return img_bgr
```

```python
import numpy as np, cv2
from .probes.compression import ela, dct_quant

from .probes.typography import font_metrics


def calibrate(authentic_paths, customer_id, out_dir="config/customers",
              target_fpr=0.01):
    ela_means, ela_stds, qhashes, jitters, sws, blockvars = [], [], [], [], [], []
    for p in authentic_paths:
        img = cv2.imread(str(p))
        if img is None: continue
        m, st = ela(img); ela_means.append(st["mean"]); ela_stds.append(st["std"])
        q = dct_quant(str(p))
        if q.get("qtable_hash"): qhashes.append(q["qtable_hash"])
        _, var, ns = noise_residual(img); blockvars.append(ns["median_block_var"])
        f = font_metrics(img)
        if f.get("status") == "ok":
            jitters.append(f["doc_baseline_jitter_px"]); sws.append(f["doc_stroke_width_px"])

    # z-threshold at the target FPR under a normal assumption over the authentic corpus
    from scipy.stats import norm
    z_thr = float(norm.ppf(1 - target_fpr))              # 1% -> 2.326
    z_thr = max(z_thr, float(np.mean(ela_means) / max(np.std(ela_means), 1e-6)) * 0.0 + z_thr)

    cal = {
        "customer_id": customer_id,
        "n_samples": len(ela_means),
        "target_fpr": target_fpr,
        "ela": {"mean": float(np.mean(ela_means)), "std": float(np.mean(ela_stds)),
                "z_threshold": round(z_thr + 1.0, 3)},     # +1.0 safety margin
        "qtable_bank": sorted(set(qhashes)),
        "noise": {"median_block_var": float(np.median(blockvars)) if blockvars else None,
                  "low_var_ratio_threshold": 0.35},
        "typography": {"baseline_jitter_px": float(np.median(jitters)) if jitters else 0.4,
                       "stroke_width_px": float(np.median(sws)) if sws else 2.0,
                       "baseline_z_threshold": 3.0, "stroke_ratio_tolerance": 0.15},
        "px_on_target_minimums": {"font_metrics": 400, "zoom": 300, "ela": 120},
    }
    d = pathlib.Path(out_dir) / customer_id; d.mkdir(parents=True, exist_ok=True)
    (d / "calibration.json").write_text(json.dumps(cal, indent=2))
    return cal

    return json.loads((pathlib.Path(out_dir) / customer_id / "calibration.json").read_text())
```

```python
"""
Grammar/schema-locked VLM client with usage tracking.
Output tokens are the dominant cost (arXiv 2607.09520). We cap them hard.
"""
import base64, json, os, time
import cv2, numpy as np
from openai import OpenAI


ADJUDICATION_SCHEMA = {
    "type": "object", "additionalProperties": False,
    "required": ["verdict", "confidence", "claims"],
    "properties": {
        "verdict": {"type": "string", "enum": [
            "tamper_detected", "authentic", "synthetic_evidence",
            "insufficient_evidence", "undetermined"]},
        "confidence": {"type": ["number", "null"], "minimum": 0, "maximum": 1},
        "claims": {"type": "array", "maxItems": 4, "items": {
            "type": "object", "additionalProperties": False,
            "required": ["id", "text", "bond"],
            "properties": {
                "id": {"type": "string"},
                "text": {"type": "string", "maxLength": 120},
                "bond": {"type": "array", "minItems": 1, "maxItems": 4,
                         "items": {"type": "string"}}}}}}}

BOND_SCHEMA = {
    "type": "object", "additionalProperties": False,
    "required": ["supports", "reason"],
    "properties": {
        "supports": {"type": "string", "enum": ["yes", "no", "insufficient_evidence"]},
        "reason": {"type": "string", "maxLength": 140}}}


class VLM:
    def __init__(self, model=None, base_url=None, api_key=None, max_tokens=120):
        self.client = OpenAI(
            base_url=base_url or os.getenv("VERDICT_BASE_URL"),
            api_key=api_key or os.getenv("VERDICT_API_KEY", "sk-noop"))
        self.model = model or os.getenv("VERDICT_MODEL", "gpt-4o-mini")
        self.max_tokens = max_tokens
        self.usage = {"calls": 0, "in": 0, "out": 0}

    @staticmethod
    def _b64(img_bgr, side=768):
        h, w = img_bgr.shape[:2]
        if max(h, w) > side:
            s = side / max(h, w)
            img_bgr = cv2.resize(img_bgr, (int(w*s), int(h*s)), interpolation=cv2.INTER_AREA)
        ok, buf = cv2.imencode(".jpg", img_bgr, [cv2.IMWRITE_JPEG_QUALITY, 92])
        return base64.b64encode(buf).decode()

    def ask(self, prompt, images=None, schema=None, max_tokens=None) -> dict:
        content = [{"type": "text", "text": prompt}]
        for im in (images or []):
            content.append({"type": "image_url",
                            "image_url": {"url": "data:image/jpeg;base64," + self._b64(im)}})
        kw = {}
        if schema:
            kw["response_format"] = {"type": "json_schema", "json_schema": {
                "name": "out", "strict": True, "schema": schema}}
        t0 = time.time()
        r = self.client.chat.completions.create(
            model=self.model, messages=[{"role": "user", "content": content}],
            max_tokens=max_tokens or self.max_tokens, temperature=0.1, **kw)
        u = r.usage
        self.usage["calls"] += 1
        self.usage["in"] += u.prompt_tokens
        self.usage["out"] += u.completion_tokens
        out = json.loads(r.choices[0].message.content)
        out["_meta"] = {"latency_s": round(time.time()-t0, 3),
                        "in_tokens": u.prompt_tokens, "out_tokens": u.completion_tokens}
        return out

    def fresh(self):
        """A context-free instance. The Verifier MUST use this."""
        return VLM(self.model, str(self.client.base_url), self.client.api_key, self.max_tokens)
```

```python
import cv2, pytesseract
from ..probes.cheap import exif_audit, phash, hamming, cross_field


class Triage:
    def __init__(self, ledger, cal, phash_index=None):
        self.ledger, self.cal = ledger, cal
        self.index = phash_index if phash_index is not None else {}

    def run(self, claim_id, img_path):
        img = cv2.imread(str(img_path))
        ex = exif_audit(img_path)
        ph = phash(img)
        dup = [(cid, d) for cid, h in self.index.items()
               if (d := hamming(ph, h)) <= 6 and cid != claim_id]
        ocr = pytesseract.image_to_string(cv2.cvtColor(img, cv2.COLOR_BGR2GRAY))
        cf = cross_field(ocr)

        flags, score = [], 0.0
        if dup:                          flags.append(f"duplicate_of:{dup[0][0]}"); score += 0.9
        if cf["status"] == "mismatch":   flags.append("arithmetic_mismatch");        score += 0.7
        for f in ex["flags"]:             flags.append(f); score += 0.25
        self.index[claim_id] = ph

        return {"agent": "triage", "vlm_calls": 0, "phash": ph, "exif": ex,
                "cross_field": cf, "duplicates": dup, "flags": flags,
                "risk": round(min(score, 1.0), 3), "ocr_text": ocr,
                "decision": "escalate" if score >= 0.25 else "low_risk"}
```

```python
"""
PROBER: pick the probe maximising expected info-gain per unit cost, under a budget.
Plans, but has NO authority to decide.
"""
import numpy as np, cv2
from ..probes.geometric import zoom, enhance
from ..probes.compression import ela, ela_region_z, dct_quant, jpeg_ghost
from ..probes.noise import noise_residual, region_hf_ratio, copy_move
from ..probes.typography import font_metrics

COST = {"zoom": 1, "ela": 1, "dct_quant": 1, "noise_residual": 2,
        "font_metrics": 1, "copy_move": 2, "enhance": 1, "jpeg_ghost": 2}


class Prober:
    def __init__(self, ledger, cal, budget=8):
        self.ledger, self.cal, self.budget = ledger, cal, budget

    def run(self, claim_id, img_path, triage, target_bbox=None):
        img = cv2.imread(str(img_path))
        H, W = img.shape[:2]
        spent, n, cards = 0, 0, []
        bbox = target_bbox or self._guess_bbox(img, triage)

        def emit(probe, params, bb, crop, obs, num, cost):
            nonlocal spent, n
            n += 1; spent += cost
            c = self.ledger.add(claim_id, f"EV-{n}", probe, params, bb, crop, obs, num, cost)
            cards.append(c); return c

        # --- P0 zoom: ALWAYS first. Nothing can be judged at 34 px. ---
        crop, zi = zoom(img, bbox)
        if crop is not None:
            emit("zoom", {"pad": 0.10}, zi["bbox"], crop,
                 f"native-resolution crop of the target field",
                 {"px_on_target": zi["px_on_target"]}, COST["zoom"])

        # --- P1 ELA, z-scored against THIS customer's calibration ---
        emap, est = ela(img)
        z = ela_region_z(emap, bbox, self.cal["ela"]["mean"], self.cal["ela"]["std"])
        thr = self.cal["ela"]["z_threshold"]
        vis = cv2.applyColorMap(np.clip(emap*8, 0, 255).astype(np.uint8), cv2.COLORMAP_INFERNO)
        emit("ela", {"qualities": [70, 80, 90]}, bbox,
             vis[int(bbox[1]):int(bbox[3]), int(bbox[0]):int(bbox[2])],
             f"ELA residual z={z['z']} vs calibrated threshold {thr}",
             {"z": z["z"], "threshold": thr,
              "exceeds": bool(z["z"] > thr)}, COST["ela"])

        # --- P2 quantization tables: qtable-bank membership + double compression ---
        q = dct_quant(str(img_path))
        in_bank = q.get("qtable_hash") in self.cal.get("qtable_bank", [])
        emit("dct_quant", {}, None, crop if crop is not None else img,
             f"qtable {'IN' if in_bank else 'NOT IN'} customer bank",
             {"in_qtable_bank": bool(in_bank), "n_tables": q.get("n_tables", 0)},
             COST["dct_quant"])

        # --- P3 typography (only if we actually have the pixels) ---
        if spent < self.budget and zi["px_on_target"] >= self.cal["px_on_target_minimums"]["font_metrics"]:
            f = font_metrics(img, bbox)
            if f.get("status") == "ok" and "baseline_z" in f:
                emit("font_metrics", {"bbox": bbox}, bbox, crop,
                     f"baseline offset {f['baseline_offset_px']}px "
                     f"(doc jitter {f['doc_baseline_jitter_px']}px), "
                     f"stroke ratio {f.get('stroke_ratio')}",
                     {"baseline_z": f["baseline_z"],
                      "stroke_ratio": f.get("stroke_ratio", 1.0),
                      "threshold": self.cal["typography"]["baseline_z_threshold"]},
                     COST["font_metrics"])

        # --- P4 noise residual: the anti-diffusion probe ---
        if spent < self.budget:
            _, var, ns = noise_residual(img)
            hf = region_hf_ratio(var, bbox, img.shape)
            emit("noise_residual", {}, bbox, crop if crop is not None else img,
                 f"region high-frequency variance is {hf['hf_ratio']}x the document median",
                 {"hf_ratio": hf["hf_ratio"],
                  "suspicious_if_below": self.cal["noise"]["low_var_ratio_threshold"]},
                 COST["noise_residual"])

        # --- P5 copy-move, only if triage hinted at duplication ---
        if spent < self.budget and "duplicate" in " ".join(triage.get("flags", [])):
            pairs, ci = copy_move(img)
            emit("copy_move", {}, None, img,
                 f"{ci['n_pairs']} self-similar keypoint pairs beyond 40px",
                 {"n_pairs": ci["n_pairs"], "frac": ci["frac_of_kp"]}, COST["copy_move"])

        return {"agent": "prober", "cards": cards, "budget_spent": spent,
                "target_bbox": bbox}

    def _guess_bbox(self, img, triage):
        """Prefer the field named by the cross-field mismatch; else the largest number."""
        from ..probes.cheap import NUM
        import pytesseract
        from pytesseract import Output
        d = pytesseract.image_to_data(cv2.cvtColor(img, cv2.COLOR_BGR2GRAY),
                                      output_type=Output.DICT)
        best, area = None, 0
        for i, t in enumerate(d["text"]):
            s = t.strip().replace(",", "").replace(".", "")
            if s.isdigit() and len(s) >= 3 and int(d["conf"][i]) > 40:
                a = d["width"][i] * d["height"][i]
                if a > area:
                    area, best = a, [d["left"][i], d["top"][i],
                                     d["left"][i]+d["width"][i], d["top"][i]+d["height"][i]]
        H, W = img.shape[:2]
        return best or [W//4, H//2, 3*W//4, H//2 + H//12]
```

`verdict/agents/adjudicator.py`:

```python
"""
ADJUDICATOR -- 🔒 PERCEPTUALLY STARVED.

It receives a TEXT MANIFEST of evidence cards. It NEVER receives the image.
Not downsampled. Not a thumbnail. Never.

Operationalises Starve to Perceive (arXiv 2605.18603) at inference time:
"If a model can succeed without actively looking, it will never learn to look."
Here, looking is not encouraged -- it is the only available channel.
"""
from ..vlm import ADJUDICATION_SCHEMA
from ..types import Claim

PROMPT = """You are a forensic adjudicator. You CANNOT see the document.
You may reason ONLY from the evidence manifest below, which was produced by
deterministic forensic probes.

Rules:
- Every claim MUST cite the evidence card ids it is derived from, in "bond".
- Do NOT claim anything the cited cards do not directly support.
- If the evidence is insufficient, return verdict "insufficient_evidence".
- Never infer visual properties (colour, shadow, texture, layout) unless a card
  explicitly reports them.

EVIDENCE MANIFEST
{manifest}

CALIBRATION (this customer's authentic baseline)
{calibration}
"""


class Adjudicator:
    def __init__(self, vlm, cal):
        self.vlm, self.cal = vlm, cal

    def run(self, cards):
        manifest = "\n".join(c.manifest_line() for c in cards)
        cal = (f"ela_z_threshold={self.cal['ela']['z_threshold']}  "
               f"baseline_z_threshold={self.cal['typography']['baseline_z_threshold']}  "
               f"noise_low_var_ratio={self.cal['noise']['low_var_ratio_threshold']}")
        out = self.vlm.ask(PROMPT.format(manifest=manifest, calibration=cal),
                           images=None,                 # <<<<<< STARVATION ENFORCED HERE
                           schema=ADJUDICATION_SCHEMA, max_tokens=120)
        valid = {c.id for c in cards}
        claims = []
        for c in out.get("claims", []):
            bond = [b for b in c["bond"] if b in valid]   # drop hallucinated card ids
            if bond:
```

## 9. Verifier Agent (bonds)

`verdict/agents/verifier.py`:

```python
"""
VERIFIER -- 🔒 fresh model instance, NO shared context.

Re-runs each claim against ONLY its cited crops, everything else blacked out.
If the claim cannot be independently reproduced from its own evidence, the bond
BREAKS and the claim is struck -- even if it happens to be true.

This is the runtime gate for wrong-evidence-right-answer (arXiv 2607.19793)
and evidence misalignment (arXiv 2605.12571).
"""
import numpy as np, cv2
from ..vlm import BOND_SCHEMA

PROMPT = """Below is a forensic finding and the ONLY evidence cited for it.
Everything not cited has been blacked out.

FINDING: "{claim}"

Numeric readings from the cited probes:
{numeric}

Does this evidence, ON ITS OWN, support the finding?
Answer "yes", "no", or "insufficient_evidence". Be strict: if the evidence is of
the wrong TYPE to assess this finding, answer "insufficient_evidence"."""


class Verifier:
    def __init__(self, vlm, ledger):
        self.vlm, self.ledger = vlm.fresh(), ledger        # fresh() == no shared context

    def test(self, claim, cards_by_id):
        # 1. provenance -- every cited card must exist and still hash correctly
        for cid in claim.bond:
            if cid not in cards_by_id or not self.ledger.verify_provenance(cid):
                claim.bond_status = "broken"
                claim.bond_detail = {"reason": f"provenance failure on {cid}"}
                return claim

        cited = [cards_by_id[c] for c in claim.bond]
        imgs = [cv2.imread(c.crop_path) for c in cited if c.crop_path]
        imgs = [i for i in imgs if i is not None]
        numeric = "\n".join(f"{c.id} {c.probe}: {c.numeric}" for c in cited)

        r = self.vlm.ask(PROMPT.format(claim=claim.text, numeric=numeric),
                         images=imgs, schema=BOND_SCHEMA, max_tokens=60)
        holds = r["supports"] == "yes"
        claim.bond_status = "holds" if holds else "broken"
        claim.bond_detail = {"verifier": r["supports"], "reason": r["reason"],
                             "cited": claim.bond, "out_tokens": r["_meta"]["out_tokens"]}
        return claim

    def test_all(self, claims, cards):
        by_id = {c.id: c for c in cards}
        return [self.test(c, by_id) for c in claims]
```

## 10. Control Agent

`verdict/agents/control.py`:

```python
"""
CONTROLS -- adversarial self-tests, run on EVERY verdict.

  blank    grey crop, same prompt        -> a finding here = prompt hallucination
  shuffle  cited crop -> random crop     -> survival = LAZY PERCEPTION
  ablate   strip filename/EXIF/OCR text  -> confidence drop = modality shortcut
"""
import numpy as np, cv2
from ..vlm import BOND_SCHEMA
from .verifier import PROMPT as BOND_PROMPT


class Control:
    def __init__(self, vlm, ledger):
        self.vlm, self.ledger = vlm.fresh(), ledger

    def blank(self, claim, cards_by_id):
        grey = np.full((448, 448, 3), 128, np.uint8)
        r = self.vlm.ask(BOND_PROMPT.format(claim=claim.text, numeric="(withheld)"),
                         images=[grey], schema=BOND_SCHEMA, max_tokens=60)
        return {"control": "blank", "result": r["supports"],
                "pass": r["supports"] != "yes"}

    def shuffle(self, claim, cards_by_id, full_img):
        """THE sharpest test. Swap the evidence for a random crop of equal size."""
        cited = [cards_by_id[c] for c in claim.bond if c in cards_by_id]
        H, W = full_img.shape[:2]
        fakes = []
        for c in cited:
            real = cv2.imread(c.crop_path) if c.crop_path else None
            h, w = (real.shape[:2] if real is not None else (200, 200))
            h, w = min(h, H-1), min(w, W-1)
            y, x = np.random.randint(0, H-h), np.random.randint(0, W-w)
            fakes.append(full_img[y:y+h, x:x+w])
        numeric = "\n".join(f"{c.id} {c.probe}: {c.numeric}" for c in cited)
        r = self.vlm.ask(BOND_PROMPT.format(claim=claim.text, numeric=numeric),
                         images=fakes, schema=BOND_SCHEMA, max_tokens=60)
        survived = r["supports"] == "yes"
        return {"control": "shuffle", "result": r["supports"],
                "pass": not survived,
                "note": "LAZY PERCEPTION: claim survived evidence replacement"
                        if survived else "claim depended on its evidence"}

    def metadata_ablation(self, rerun_fn):
        """Re-run the whole pipeline with EXIF/filename/OCR text stripped."""
        base, ablated = rerun_fn(strip=False), rerun_fn(strip=True)
        drop = (base or 0) - (ablated or 0)
        return {"control": "metadata_ablation",
                "conf_full": base, "conf_ablated": ablated, "drop": round(drop, 3),
                "pass": drop < 0.15,
                "note": "MODALITY SHORTCUT -- report the ablated confidence"
                        if drop >= 0.15 else "not metadata-dependent"}
```

`verdict/orchestrator.py`:

```python
"""Wires the five agents together. Enforces the separation of powers."""
import time, cv2
from .ledger import Ledger
from .calibrate import load as load_cal
from .vlm import VLM
from .agents.triage import Triage
from .agents.prober import Prober
from .agents.adjudicator import Adjudicator
from .agents.verifier import Verifier
from .agents.control import Control
from .types import Verdict

PRICE_IN, PRICE_OUT = 3e-6, 15e-6          # $/token -- set to your provider's rates


class Orchestrator:
    def __init__(self, customer_id="C-0041", budget=8, run_controls=True,
                 starve=True, use_probes=True):
        self.cal = load_cal(customer_id)
        self.ledger = Ledger()
        self.vlm = VLM()
        self.triage = Triage(self.ledger, self.cal)
        self.prober = Prober(self.ledger, self.cal, budget=budget)
        self.adj = Adjudicator(self.vlm, self.cal)
        self.ver = Verifier(self.vlm, self.ledger)
        self.ctl = Control(self.vlm, self.ledger)
        self.run_controls, self.starve, self.use_probes = run_controls, starve, use_probes

    def run(self, claim_id, img_path):
        t0 = time.time()
        self.vlm.usage = {"calls": 0, "in": 0, "out": 0}
        img = cv2.imread(str(img_path))
        trace = {"claim_id": claim_id, "steps": []}

        # 1. TRIAGE -- zero VLM calls
        tri = self.triage.run(claim_id, img_path)
        trace["steps"].append(tri)
        if tri["decision"] == "low_risk" and not tri["flags"]:
            return self._finish(claim_id, "authentic", 0.75, [], {}, t0, [], trace)

        # 2. PROBER
        if self.use_probes:
            pr = self.prober.run(claim_id, img_path, tri)
            cards = pr["cards"]; trace["steps"].append(
                {"agent": "prober", "n_cards": len(cards), "spent": pr["budget_spent"]})
        else:                                        # ABLATION: no probes
            crop = cv2.resize(img, (448, 448))
            cards = [self.ledger.add(claim_id, "EV-1", "downsample", {}, None, crop,
                                     "whole document at 448x448", {}, 0)]

        # 3. ADJUDICATOR (starved unless ablated)
        if self.starve:
            a = self.adj.run(cards)
        else:                                        # ABLATION: adjudicator sees the image
            a = self.adj.run(cards); a["_unstarved"] = True
            a2 = self.vlm.ask("Assess this document for tampering.", images=[img],
                              schema=__import__("verdict.vlm", fromlist=["x"]).ADJUDICATION_SCHEMA,
                              max_tokens=120)
            a["verdict"], a["confidence"] = a2["verdict"], a2.get("confidence")
        trace["steps"].append({"agent": "adjudicator", "verdict": a["verdict"],
                               "n_claims": len(a["claims"]), "out_tokens": a["meta"]["out_tokens"]})

        # 4. VERIFIER -- bonds
        claims = self.ver.test_all(a["claims"], cards)
        broken = [c for c in claims if c.bond_status == "broken"]
        trace["steps"].append({"agent": "verifier",
                               "holds": len(claims) - len(broken), "broken": len(broken),
                               "details": [c.bond_detail for c in broken]})

        # 5. CONTROLS
        controls = {}
        if self.run_controls and claims:
            by_id = {c.id: c for c in cards}
            live = [c for c in claims if c.bond_status == "holds"] or claims
            controls["blank"] = self.ctl.blank(live[0], by_id)
            controls["shuffle"] = self.ctl.shuffle(live[0], by_id, img)
            trace["steps"].append({"agent": "control", **controls})

        verdict = a["verdict"]
        conf = a.get("confidence")
        if not [c for c in claims if c.bond_status == "holds"]:
            verdict, conf = "undetermined", None       # nothing survived -> no verdict
        if controls.get("shuffle", {}).get("pass") is False:
            conf = min(conf or 0, 0.5)                 # lazy perception -> downgrade

        return self._finish(claim_id, verdict, conf, claims, controls, t0,
                            [c.id for c in cards], trace)

    def _finish(self, cid, verdict, conf, claims, controls, t0, chain, trace):
        u = self.vlm.usage
        cost = {"vlm_calls": u["calls"], "in_tokens": u["in"], "out_tokens": u["out"],
                "usd": round(u["in"]*PRICE_IN + u["out"]*PRICE_OUT, 6)}
        v = Verdict(cid, verdict, conf, claims, controls, cost,
                    round(time.time()-t0, 3), chain)
        v_d = v.to_dict(); v_d["trace"] = trace
        return v_d
```

```python
"""Render a verdict as a single report-card PNG. Faster than a web UI and it films better."""
import cv2, numpy as np

OK, BAD, WARN, DIM = (110,224,138), (122,122,255), (121,212,255), (150,150,150)

def render(img_bgr, verdict: dict, cards: list, out_path: str, W=1600, H=900):
    canvas = np.full((H, W, 3), 18, np.uint8)

    # ---- left: document with probe bboxes ----
    h, w = img_bgr.shape[:2]
    s = min((H-120)/h, (W//2-40)/w)
    doc = cv2.resize(img_bgr, (int(w*s), int(h*s)))
    dh, dw = doc.shape[:2]
    for c in cards:
        if c.bbox:
            x1,y1,x2,y2 = [int(v*s) for v in c.bbox]
            cv2.rectangle(doc, (x1,y1), (x2,y2), WARN, 2)
            cv2.putText(doc, c.id, (x1, max(12,y1-5)), cv2.FONT_HERSHEY_SIMPLEX, .5, WARN, 1)
    canvas[70:70+dh, 20:20+dw] = doc

    def txt(y, s_, col=(230,230,230), sc=.55, x=W//2+20, th=1):
        cv2.putText(canvas, s_, (x, y), cv2.FONT_HERSHEY_SIMPLEX, sc, col, th, cv2.LINE_AA)

    txt(45, "V E R D I C T   ·   EVIDENCE-BONDED REASONING", (255,224,110), .7, 20)
    y = 90
    txt(y, "PROBE TRACE", DIM, .5); y += 28
    for c in cards:
        n = " ".join(f"{k}={v}" for k, v in list(c.numeric.items())[:3])
        txt(y, f"{c.id:<6}{c.probe:<12} px={c.px_on_target:<6} {n}"); y += 22

    y += 18; txt(y, "BONDED CLAIMS", DIM, .5); y += 28
    for cl in verdict["claims"]:
        held = cl["bond_status"] == "holds"
        txt(y, f"{'OK ' if held else 'XX '} {cl['id']}  {cl['text'][:58]}",
            OK if held else BAD); y += 20
        txt(y, f"       bond {cl['bond']}" + ("" if held else
            f"  BROKEN: {cl['bond_detail'].get('reason','')[:44]}"), DIM, .45); y += 24

    y += 14
    ctl = verdict.get("controls", {})
    sh = ctl.get("shuffle", {})
    txt(y, f"CONTROLS   shuffle={'PASS' if sh.get('pass') else 'FAIL'}",
        OK if sh.get("pass") else BAD, .5); y += 26
    cost = verdict["cost"]
    txt(y, f"COST  {cost['vlm_calls']} calls · {cost['out_tokens']} out-tokens · "
           f"${cost['usd']:.4f} · {verdict['latency_s']}s", (255,224,110), .5)

    v = verdict["verdict"]
    col = BAD if v in ("tamper_detected","synthetic_evidence") else (
          OK if v == "authentic" else WARN)
    cv2.rectangle(canvas, (W//2+16, H-70), (W-20, H-24), col, 2)
    txt(H-40, v.upper().replace("_"," "), col, .8, W//2+30, 2)

    cv2.imwrite(out_path, canvas)
    return out_path
```

## 16.5 The physical element — do not skip this

**The challenge format is a *Physical Prototype Challenge*.** The brief tells judges to score "design, build, test and demonstrate working prototypes." A pure-software entry starts at a structural disadvantage on that criterion regardless of software quality.

### Build: the VERDICT Capture Station — ~₹1,000, 30 minutes

| Part | ₹ |
|---|---|
| Phone/webcam clamp or gooseneck stand | 400 |
| USB LED light or small ring light | 200 |
| A4 foam-board / plywood base plate | 100 |
| **Printed fiducial marker** (ArUco 4×4) + **18% grey card** + **mm ruler**, taped to the plate | 50 (print) |
| USB cable, tape, clips | 150 |

```
        ┌── clamp ──┐
        │  [PHONE]  │   ← fixed height, fixed angle
        │     │     │
     [LED]    │    [LED]  ← fixed, diffuse, known colour temp
              ▼
   ╔═══════════════════════╗
   ║ ▣            ▣        ║  ▣ = ArUco fiducials (known mm spacing)
   ║     ┌─────────┐       ║
   ║     │ RECEIPT │       ║
   ║     └─────────┘       ║
   ║ ▣  [GREY CARD]  ▣     ║  ← per-capture white balance reference
   ║ |‗‗|‗‗|‗‗|‗‗| mm      ║
   ╚═══════════════════════╝
```

### Why it is load-bearing, not decorative

This matters — a bolted-on gadget scores worse than none at all. Three genuine dependencies:

1. **It answers VERDICT's own refusal path.** idea.md §13 has the system refuse and ask for *"the original file, not a screenshot or forward, ≥1500 px across the total field."* The capture station **is that answer.** It produces originals with intact EXIF and known provenance, instead of thrice-forwarded WhatsApp screenshots with the compression history destroyed.

2. **The fiducial makes `px_on_target` physically meaningful.** With ArUco markers at known mm spacing, the system computes **pixels per millimetre**, so the resolution gate stops being an arbitrary pixel count and becomes a physical measurement: *"this probe requires 12 px/mm; you have provided 4."* That is a far stronger claim than a bare threshold, and it is only possible with a physical reference in frame.

3. **Fixed geometry collapses calibration variance.** The N=10 calibration in §4 gets dramatically tighter when lighting, distance and angle are constant. The grey card gives a per-capture white-balance reference, so ELA and noise statistics become comparable *across* captures rather than only within one. You can show this directly: calibration std with the rig versus handheld.

### Build it at hour 12, not hour 0

It takes 30 minutes and needs no debugging. Assemble it while the ablations are running, then shoot the demo with a physical forged receipt placed under the rig.

**Bonus demo shot:** place a forged printed receipt on the plate → capture → catch it live, on camera, end to end. That single 20-second clip converts "software project" into "working system."
